\begin{frame}{1. El problema}
  \ponente{Juan Diego Canaza Paucara}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Las heladas son el \textbf{principal peligro agroclimático del altiplano peruano}:
      \begin{itemize}\itemsep0.3em
        \item Reducen el rendimiento de cultivos altoandinos (papa, quinua, cañihua).
        \item Comprometen la seguridad alimentaria de las familias productoras.
        \item Afectan la sanidad y la productividad del ganado.
      \end{itemize}

      \vspace{0.6em}
      \alert{El vacío que atacamos:} existen registros climáticos diarios de acceso abierto, pero
      están dispersos, en formatos poco amigables y \textbf{sin instrumentos de exploración} que
      permitan responder \emph{cuándo}, \emph{dónde} y \emph{con qué intensidad} se concentra el riesgo.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Pregunta de investigación}
        \small ¿Cómo se distribuyen en el tiempo y en el territorio la frecuencia y la intensidad de
        las heladas en las trece provincias de Puno, y qué factores climáticos las explican?
      \end{block}
      \vspace{0.3em}
      \begin{block}{Producto}
        \small Un dashboard que transforma diez años de observaciones diarias en indicadores,
        visualizaciones y hallazgos accionables.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

    Producto      & ERA5-Land / ERA5 (\emph{era5\_seamless}) & MERRA-2 (NASA POWER) \\
    Institución   & ECMWF — Copernicus (C3S) & NASA Langley Research Center \\
    Resolución    & \textbf{0,1° $\approx$ 9 km} & 0,5° $\times$ 0,625° $\approx$ 55 km \\
    Uso           & Alimenta todo el dashboard & Sólo validación cruzada \\
    \bottomrule
  \end{tabular}

  \vspace{0.8em}
  \small
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Estructura del dataset}
      \begin{itemize}\itemsep0.2em
        \item \textbf{47\,489} registros (3\,653 días $\times$ 13 provincias)
        \item \textbf{52} variables: 31 numéricas, 20 categóricas
        \item Variable temporal (diaria) y geográfica (lat/lon/altitud)
        \item Sin datos personales ni sensibles
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Por qué descartamos la malla gruesa}
        \scriptsize Con MERRA-2, \textbf{Chucuito, El Collao y Yunguyo caían en la misma celda} y
        recibían series \emph{idénticas}; además asignaba 3\,668 msnm a Sandia, cuya capital está a
        2\,170 msnm. Cualquier ranking provincial habría sido un artefacto de la malla.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{4. Limpieza, transformación e integración}
  \ponente{Juan Diego Canaza Paucara}
  \scriptsize
  Pipeline auditable de \textbf{13 etapas}; cada una registra qué hizo y a cuántos registros afectó.
  \vspace{0.4em}

  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{enumerate}\itemsep0.08em
        \item Normalización del esquema (19 variables)
        \item Fecha ISO $\rightarrow$ tipo temporal
        \item \alert{Homogeneización de unidades} (km/h$\rightarrow$m/s, hPa$\rightarrow$kPa, s$\rightarrow$h)
        \item Centinelas de ausencia $\rightarrow$ nulos
        \item Validación de rangos físicos
        \item Coherencia térmica (mín $\leq$ media $\leq$ máx)
        \item Marcado de precipitación implausible
      \end{enumerate}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{enumerate}\itemsep0.08em
        \setcounter{enumi}{7}
        \item Unicidad de la clave (provincia, fecha)
        \item Reindexado del calendario completo
        \item Imputación temporal por provincia
        \item \alert{Integración de la fuente secundaria}
        \item 23 variables derivadas
        \item Tipado eficiente (6,8 MB en memoria)
      \end{enumerate}
    \end{column}
  \end{columns}

  \vspace{0.5em}
  \begin{block}{Un hallazgo del propio proceso de limpieza}
    \scriptsize Un umbral mal calibrado en la radiación solar (MJ/m²/día, no kWh/m²/día) habría anulado
    \textbf{46\,564 observaciones} que la imputación después rellenaría por interpolación,
    corrompiendo el análisis multivariante \emph{sin dejar rastro visible}. Verificar las unidades
    reales del servicio —y no las que sugiere el nombre del parámetro— fue decisivo.
  \end{block}
\end{frame}

\begin{frame}{5. Arquitectura del software}
  \ponente{Angie Tatiana Luque Pacheco}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{Separación estricta por capas}
      \begin{itemize}\itemsep0.25em
        \item \texttt{config/} — única fuente de verdad: rutas, metadatos, dominio, identidad visual.
        \item \texttt{utils/} — lógica de dominio. \alert{No importa Streamlit}: se reutiliza desde el
              cuaderno, los scripts y las pruebas.
        \item \texttt{components/} — componentes visuales reutilizables.
        \item \texttt{pages/} — sólo orquestan componentes; sin HTML ni reglas de estilo.
        \item \texttt{scripts/} — flujo reproducible 01$\rightarrow$04.
      \end{itemize}
      \vspace{0.4em}
      PEP 8, \emph{type hints} y docstrings estilo Google en todo el código.
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{block}{Decisiones de UX/UI}
        \scriptsize
        \textbf{Color en dos niveles.} La identidad es pastel, pero un pastel puro no sostiene la
        identidad de serie en un gráfico. Por eso:
        \begin{itemize}\itemsep0.1em
          \item \textbf{Superficie} (fondos, tarjetas, insignias): pastel.
          \item \textbf{Tinta de datos}: mismos tonos, pasos verificados.
        \end{itemize}
        \vspace{0.3em}
        Ninguna paleta se eligió a ojo: se obtuvieron por \textbf{optimización con restricciones} en el
        espacio OKLCH —minimizando saturación sujeto a superar las pruebas de contraste y de
        separación cromática bajo protanopía y deuteranopía—.
        \vspace{0.25em}
        Tope estructural de \textbf{4 series} en dispersión y mapas: más allá, la distinción no está
        garantizada.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

        registran 0 y 930 días respectivamente. La diferencia es la \textbf{inercia térmica del lago
        Titicaca}, que atenúa el enfriamiento nocturno en las provincias circunlacustres.
        \vspace{0.2em}

        \alert{Consecuencia práctica:} zonificar sólo por altitud sería un error de diseño de política.
      \end{block}
    \end{column}
  \end{columns}

\begin{frame}{7. Resultados — tendencia y técnicas avanzadas (P3, P5, P6)}
  \ponente{Jhon Marco Aracayo Mamani}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Tendencia interanual (P3)} — se aplican \emph{dos} métodos, no uno:
      \begin{itemize}\itemsep0.15em
        \item \textbf{OLS}: óptimo con residuos normales, sensible a atípicos.
        \item \textbf{Mann-Kendall + pendiente de Sen}: no paramétrico, resistente.
      \end{itemize}
      \vspace{0.25em}
      Reportar ambos es lo honesto: si coinciden, la conclusión es robusta; si discrepan,
      \textbf{la discrepancia es el hallazgo} y se declara en lugar de elegir el método más favorable.

      \vspace{0.6em}
      \textbf{Validación cruzada entre fuentes}
      \begin{itemize}\itemsep0.15em
        \item $r = 0{,}611$ ; sesgo medio $+1{,}29$ °C
        \item Concordancia en la clasificación de helada: \textbf{83,1\,\%}
        \item La correlación moderada es \emph{coherente} con la diferencia de resolución
              (9 km frente a 55 km), no un error.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Técnicas avanzadas aplicadas (P5, P6)}
      \begin{itemize}\itemsep0.12em
        \item \textbf{PCA} — factores latentes del sistema; \emph{scree}, cargas, biplot y 3D.
        \item \textbf{K-Means} — zonificación agroclimática; $k$ por codo y silueta.
        \item \textbf{Jerárquico (Ward)} — dendrograma de las 13 provincias.
        \item \textbf{Isolation Forest} — días atípicos \emph{multivariantes}.
        \item \textbf{Z modificada (MAD)} — extremos robustos a contaminación.
        \item \textbf{Random Forest} — importancia por permutación, ROC y PR.
        \item \textbf{Correlación} Pearson/Spearman con valores $p$ y VIF.
        \item \textbf{Mapa geográfico interactivo}.
      \end{itemize}
      \vspace{0.3em}
      {\tiny\textcolor{pnInkSoft}{Partición cronológica, no aleatoria: con series autocorrelacionadas,
      una partición aleatoria filtraría información y inflaría las métricas.}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{8. Conclusiones}
  \ponente{Jhon Marco Aracayo Mamani}
  \small
  \begin{enumerate}\itemsep0.35em
    \item El riesgo por heladas en Puno es \textbf{estructuralmente heterogéneo}: la brecha entre la
          provincia más y la menos expuesta abarca todo el rango posible.
    \item El umbral agronómico (3 °C) multiplica por \textbf{2,5} los días de riesgo respecto al
          meteorológico: la cifra relevante para el agro no es la del fenómeno físico.
    \item La \textbf{altitud es el primer factor pero no el único}: el efecto termorregulador del
          lago Titicaca domina en las provincias circunlacustres.
    \item El fenómeno se concentra en una \textbf{ventana estacional estrecha}, coincidente con la
          estación seca y el cielo despejado que permite la pérdida radiativa nocturna.
    \item Las técnicas multivariantes muestran que el sistema está gobernado por \textbf{pocos
          factores latentes} (un eje térmico y otro de humedad), lo que abarata el monitoreo.
    \item Dos reanálisis independientes \textbf{coinciden en el 83,1\,\%} de la clasificación de
          helada: la señal no es un artefacto de un producto concreto.
    \item El proyecto entrega un \textbf{instrumento reproducible} (semilla fija, pipeline
          auditable de 13 etapas, pruebas automatizadas), no sólo un conjunto de gráficos.
  \end{enumerate}
\end{frame}

\begin{frame}{8. Recomendaciones y limitaciones}
  \ponente{Jhon Marco Aracayo Mamani}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Decisiones que habilitan los datos}
      \begin{itemize}\itemsep0.18em
        \item \textbf{Gobierno regional:} focalizar el seguro agrario y la inversión en protección en
              las provincias del extremo superior del ranking, donde el beneficio marginal por unidad
              invertida es mayor.
        \item \textbf{Dir. Reg. de Agricultura:} calendarios de siembra diferenciados \emph{por
              provincia}, no regionales; alerta temprana basada en nubosidad y humedad, observables
              con antelación.
        \item \textbf{Productores:} priorizar variedades tolerantes al frío en la franja
              altitudinal crítica; concentrar medidas en la ventana estacional identificada.
        \item \textbf{Academia:} densificar la red de estaciones donde el reanálisis muestra mayor
              incertidumbre entre fuentes.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Límites que declaramos}
      \begin{itemize}\itemsep0.18em
        \item Malla de $\approx 9$ km: \textbf{suaviza extremos locales} y no resuelve fondos de valle.
        \item Reanálisis, \textbf{no estación de superficie}.
        \item \textbf{Diez años} caracterizan la variabilidad interanual, pero la OMM exige 30 para
              normales climáticas: \emph{no atribuimos cambio climático}.
        \item Una sola capital \textbf{representa} cada provincia.
        \item Sin datos de rendimiento agrícola: no cuantificamos el daño económico.
        \item El índice de riesgo tiene \textbf{pesos fijados por criterio experto}, no calibrados
              contra daño observado.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
